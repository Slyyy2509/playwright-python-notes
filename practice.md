from playwright.sync_api import sync_playwright

with sync_playwright() as p:
    browser = p.chromium.launch()
    page = browser.new_page()

    page.goto("https://playwright.dev/")
    page.wait_for_load_state("load")

    title = page.title()

    assert "Playwright" in title, (
        f"Заголовок не содержит Playwright: {title}"
    )

    print("Проверка пройдена!")
    print(f"Заголовок страницы: {title}")

    browser.close()

# 1. Цель работы
# Изучить методы поиска элементов страницы с помощью Playwright и научиться выполнять нажатия на кнопки.

# 2. Исходный код:
from playwright.sync_api import sync_playwright

with sync_playwright() as p:
    browser = p.chromium.launch()
    page = browser.new_page()

    page.set_content("""
        <button onclick="this.textContent = 'Signed in'">
            Sign in
        </button>
    """)

    page.get_by_role("button", name="Sign in").click()

    assert page.get_by_text("Signed in").is_visible()

    print("Проверка пройдена! Кнопка изменила текст.")

    browser.close()

# 3. Изученные команды
# page.set_content() — добавляет HTML-содержимое страницы.
# page.get_by_role() — ищет элемент по его роли и доступному имени.
# click() — выполняет нажатие на найденный элемент.
# get_by_text() — ищет элемент по текстовому содержимому.
# is_visible() — проверяет видимость элемента.
